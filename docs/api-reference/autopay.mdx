---
title: "Configure Autopay"
api: "POST /api/v1/autopay/schedule"
description: "Register an EIP-712 cryptographic mandate signature with the Autopay relayer to schedule automated recurring charges."
---

### Headers

<ParamField header="x-api-key" type="string" required>
  Your Arca Developer API Key.
</ParamField>

### Body Parameters

<ParamField body="subscriber" type="string" required>
  The subscriber's modular smart contract wallet address.
</ParamField>

<ParamField body="sessionPublicKey" type="string" required>
  The public key of the temporary session key pair authorized to call the Billing Gateway.
</ParamField>

<ParamField body="planId" type="string" required>
  The target plan's unique bytes32 ID.
</ParamField>

<ParamField body="tierId" type="integer" required>
  The specific pricing tier index inside the plan (e.g. `0` for Basic, `1` for Pro).
</ParamField>

<ParamField body="maxCycles" type="integer" required>
  The maximum billing cycles allowed under this signature mandate.
</ParamField>

<ParamField body="deadline" type="integer" required>
  The UNIX timestamp indicating when the session authorization signature expires.
</ParamField>

<ParamField body="signature" type="string" required>
  The EIP-712 cryptographic signature generated by the subscriber's wallet verifying the mandate.
</ParamField>

<ParamField body="buyerData" type="string">
  Optional custom metadata string associated with this subscription checkout (e.g. user ID, email).
</ParamField>

---

### Response Fields

<ResponseField name="success" type="boolean">
  Indicates whether the signature was successfully registered and scheduled.
</ResponseField>

<ResponseField name="subscriptionId" type="string">
  The unique ID of the scheduled autopay subscription mapping this user profile.
</ResponseField>

<ResponseField name="nextExecution" type="integer">
  The UNIX timestamp scheduled for the next billing execution.
</ResponseField>

<ResponseField name="status" type="string">
  The active relayer status: `active` or `pending_allowance`.
</ResponseField>

<ResponseField name="state" type="object">
  Detail state of execution on-chain:
  <Expandable title="state properties">
    <ResponseField name="currentCycle" type="integer">The current billing cycle count completed (starts at `0` if pre-authorized).</ResponseField>
    <ResponseField name="allowanceApproved" type="boolean">True if the subscriber's contract has approved the gateway USDC spender limit.</ResponseField>
    <ResponseField name="balanceSufficient" type="boolean">True if the subscriber has enough USDC to execute the initial cycle.</ResponseField>
  </Expandable>
</ResponseField>
